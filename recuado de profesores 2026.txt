<script>
    // Lista completa de grados
    const GRADOS_COMPLETOS = [];
    
    for (let letra of ['A', 'B', 'C']) {
        GRADOS_COMPLETOS.push(`Prejardin ${letra}`);
        GRADOS_COMPLETOS.push(`Jardin ${letra}`);
    }
    for (let letra of ['A', 'B', 'C', 'D', 'E', 'F', 'G', 'H']) {
        GRADOS_COMPLETOS.push(`Transicion ${letra}`);
    }
    for (let i = 1; i <= 11; i++) {
        for (let letra of ['A', 'B', 'C', 'D', 'E', 'F', 'G', 'H', 'I', 'J']) {
            GRADOS_COMPLETOS.push(`${i}${letra}`);
        }
    }
    
    // Configuración
    let META_POR_CURSO = 20000;
    const FECHA_LIMITE = new Date(2026, 4, 14); // 14 de Mayo 2026
    
    // Inicializar datos
    function inicializarDatos() {
        if (!localStorage.getItem('opalDonaciones')) {
            const donacionesIniciales = {};
            GRADOS_COMPLETOS.forEach(grado => {
                donacionesIniciales[grado] = 0;
            });
            localStorage.setItem('opalDonaciones', JSON.stringify(donacionesIniciales));
        }
        
        if (!localStorage.getItem('opalHistorial')) {
            localStorage.setItem('opalHistorial', JSON.stringify([]));
        }
        
        if (!localStorage.getItem('opalMetaPorCurso')) {
            localStorage.setItem('opalMetaPorCurso', META_POR_CURSO);
        }
        
        META_POR_CURSO = parseInt(localStorage.getItem('opalMetaPorCurso'));
        actualizarMetaMostrada();
    }
    
    function obtenerDonaciones() {
        return JSON.parse(localStorage.getItem('opalDonaciones'));
    }
    
    function guardarDonaciones(datos) {
        localStorage.setItem('opalDonaciones', JSON.stringify(datos));
    }
    
    function obtenerHistorial() {
        return JSON.parse(localStorage.getItem('opalHistorial'));
    }
    
    function guardarHistorial(historial) {
        localStorage.setItem('opalHistorial', JSON.stringify(historial));
    }
    
    function fechaLimitePasada() {
        return new Date() > FECHA_LIMITE;
    }
    
    function registrarDonacion() {
        if (fechaLimitePasada()) {
            alert("⚠️ La fecha límite (14 de Mayo) ha pasado. No se pueden registrar nuevos pagos.");
            return;
        }
        
        const nombre = document.getElementById('nombrePagador').value.trim() || "Anónimo";
        const grado = document.getElementById('gradoSelect').value;
        const monto = parseInt(document.getElementById('montoInput').value);
        
        if (!grado) {
            alert("Selecciona un curso");
            return;
        }
        
        if (!monto || monto <= 0) {
            alert("Ingresa un monto válido");
            return;
        }
        
        // Actualizar donaciones
        const donaciones = obtenerDonaciones();
        donaciones[grado] += 1;
        guardarDonaciones(donaciones);
        
        // Guardar en historial
        const historial = obtenerHistorial();
        const now = new Date();
        const fechaStr = `${now.getDate()}/${now.getMonth()+1}/${now.getFullYear()} ${now.getHours()}:${now.getMinutes()}`;
        
        historial.unshift({
            fecha: fechaStr,
            nombre: nombre,
            curso: grado,
            monto: monto,
            id: Date.now()
        });
        
        guardarHistorial(historial);
        
        // Limpiar y actualizar
        document.getElementById('montoInput').value = '';
        document.getElementById('nombrePagador').value = '';
        
        actualizarTodo();
        mostrarMensaje("✅ Pago registrado correctamente");
    }
    
    function eliminarDonacion(id) {
        let historial = obtenerHistorial();
        const donacionEliminada = historial.find(h => h.id === id);
        
        if (donacionEliminada) {
            // Restar del contador del grado
            const donaciones = obtenerDonaciones();
            donaciones[donacionEliminada.curso] = Math.max(0, donaciones[donacionEliminada.curso] - 1);
            guardarDonaciones(donaciones);
            
            // Eliminar del historial
            historial = historial.filter(h => h.id !== id);
            guardarHistorial(historial);
            
            actualizarTodo();
            mostrarMensaje("🗑️ Donación eliminada");
        }
    }
    
    function calcularRecaudado() {
        const historial = obtenerHistorial();
        return historial.reduce((total, item) => total + item.monto, 0);
    }
    
    function calcularMetaTotal() {
        return GRADOS_COMPLETOS.length * META_POR_CURSO;
    }
    
    function getGradosSinDonar() {
        const donaciones = obtenerDonaciones();
        return GRADOS_COMPLETOS.filter(grado => donaciones[grado] === 0);
    }
    
    function actualizarTodo() {
        // Recaudado
        const recaudado = calcularRecaudado();
        document.getElementById('recaudado').innerHTML = `$${recaudado.toLocaleString()}`;
        
        // Avance
        const metaTotal = calcularMetaTotal();
        const avance = metaTotal > 0 ? (recaudado / metaTotal) * 100 : 0;
        document.getElementById('avance').innerHTML = `${avance.toFixed(1)}%`;
        
        // Historial
        const historial = obtenerHistorial();
        const tbody = document.getElementById('historialBody');
        tbody.innerHTML = historial.map(item => `
            <tr>
                <td>${item.fecha}</td>
                <td>${item.nombre}</td>
                <td>${item.curso}</td>
                <td>$${item.monto.toLocaleString()}</td>
                <td><button class="btn-eliminar" onclick="eliminarDonacion(${item.id})">Eliminar</button></td>
            </tr>
        `).join('');
        
        // ¿Quién falta?
        const faltantes = getGradosSinDonar();
        const faltantesDiv = document.getElementById('listaFaltantes');
        if (faltantes.length === 0) {
            faltantesDiv.innerHTML = '<div class="sin-faltantes">🎉 ¡Todos los cursos han donado! 🎉</div>';
        } else {
            faltantesDiv.innerHTML = faltantes.map(grado => `<span class="badge-faltante">${grado}</span>`).join('');
        }
        
        // Select de grados (actualizar contadores)
        const select = document.getElementById('gradoSelect');
        if (select.options.length <= 1) {
            let options = '<option value="">Selecciona curso</option>';
            for (const grado of GRADOS_COMPLETOS) {
                options += `<option value="${grado}">${grado}</option>`;
            }
            select.innerHTML = options;
        }
        
        // Verificar fecha límite
        const advertencia = document.getElementById('advertenciaFecha');
        const btnRegistrar = document.getElementById('btnRegistrar');
        if (fechaLimitePasada()) {
            advertencia.style.display = 'block';
            btnRegistrar.disabled = true;
            btnRegistrar.style.opacity = '0.5';
        } else {
            advertencia.style.display = 'none';
            btnRegistrar.disabled = false;
            btnRegistrar.style.opacity = '1';
        }
    }
    
    function actualizarMetaMostrada() {
        const metaTotal = calcularMetaTotal();
        document.getElementById('metaMostrada').innerHTML = `$${metaTotal.toLocaleString()}`;
        actualizarTodo();
    }
    
    function editarMeta() {
        const nuevoValor = prompt("Ingrese la meta por curso (en pesos):", META_POR_CURSO);
        if (nuevoValor && !isNaN(nuevoValor) && parseInt(nuevoValor) > 0) {
            META_POR_CURSO = parseInt(nuevoValor);
            localStorage.setItem('opalMetaPorCurso', META_POR_CURSO);
            actualizarMetaMostrada();
            mostrarMensaje(`✅ Meta actualizada: $${META_POR_CURSO.toLocaleString()} por curso`);
        }
    }
    
    function descargarExcel() {
        const donaciones = obtenerDonaciones();
        const datosExcel = GRADOS_COMPLETOS.map(grado => ({
            'Grado': grado,
            'Cantidad de Pagos': donaciones[grado] || 0
        }));
        
        const ws = XLSX.utils.json_to_sheet(datosExcel);
        const wb = XLSX.utils.book_new();
        XLSX.utils.book_append_sheet(wb, ws, 'Donaciones');
        XLSX.writeFile(wb, `opal_donaciones_${new Date().toISOString().slice(0,10)}.xlsx`);
        mostrarMensaje("📎 Excel descargado");
    }
    
    function mostrarMensaje(msg) {
        // Simple alert por ahora
        console.log(msg);
    }
    
    // QR Image
    document.getElementById('qrInput').addEventListener('change', function(e) {
        const file = e.target.files[0];
        if (file) {
            const reader = new FileReader();
            reader.onload = function(event) {
                const img = document.getElementById('qrImagen');
                const placeholder = document.getElementById('qrPlaceholder');
                img.src = event.target.result;
                img.style.display = 'block';
                placeholder.style.display = 'none';
                localStorage.setItem('opalQR', event.target.result);
            };
            reader.readAsDataURL(file);
        }
    });
    
    function cargarQRGuardado() {
        const qrGuardado = localStorage.getItem('opalQR');
        if (qrGuardado) {
            const img = document.getElementById('qrImagen');
            const placeholder = document.getElementById('qrPlaceholder');
            img.src = qrGuardado;
            img.style.display = 'block';
            placeholder.style.display = 'none';
        }
    }
    
    // Inicializar
    inicializarDatos();
    cargarQRGuardado();
    actualizarTodo();
</script>
